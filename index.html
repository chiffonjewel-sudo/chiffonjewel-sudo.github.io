const CACHE_KEY = 'cherrynames_cache'; const CACHE_TTL = 15 * 60 * 1000; // 15 minutes async function fetchLiveDrops() { // Check cache const cached = localStorage.getItem(CACHE_KEY); if (cached) { const parsed = JSON.parse(cached); if (Date.now() - parsed.timestamp < CACHE_TTL) { return parsed.drops; } } // Replace with YOUR Cloudflare Worker URL const WORKER_URL = 'https://namemc-proxy.YOURNAME.workers.dev'; const target = 'https://namemc.com/minecraft-names?sort=asc&length_op=eq&length=3'; try { const res = await fetch(`${WORKER_URL}?url=${encodeURIComponent(target)}`); const html = await res.text(); // Parse const parser = new DOMParser(); const doc = parser.parseFromString(html, 'text/html'); if (doc.title.includes('moment') || doc.title.includes('Attention')) { throw new Error('Cloudflare blocked request'); } const drops = []; doc.querySelectorAll('table tbody tr').forEach(row => { const name = row.querySelector('a')?.textContent.trim(); const time = row.querySelector('time')?.getAttribute('datetime'); if (name) drops.push({ name, dropTime: time || 'Unknown', status: 'later' }); }); // Cache localStorage.setItem(CACHE_KEY, JSON.stringify({ drops, timestamp: Date.now() })); return drops; } catch (err) { console.warn('Live fetch failed, using fallback:', err.message); return null; // Fall back to mockData } } // Initialize with fallback, then try live renderUpcomingDrops(mockData.upcomingDrops); fetchLiveDrops().then(live => { if (live && live.length > 0) { renderUpcomingDrops(live); } });
